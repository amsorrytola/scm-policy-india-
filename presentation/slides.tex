% ── SLIDE 3: Motivation ─────────────────────────────────────────────────────
\begin{frame}{Motivation --- Why Study Bihar's Prohibition?}
  \vspace{0.15cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      \begin{tcolorbox}[amberbox]
        \small\textbf{The policy.} Effective \highlight{April 5, 2016}, Bihar
        enacted total prohibition on the sale and consumption of alcohol ---
        the largest sub-national prohibition by population in modern history
        ($\sim$120M people).
      \end{tcolorbox}
      \vspace{0.3cm}
      \textbf{\navy{Why this case is methodologically interesting:}}
      \begin{itemize}\setlength\itemsep{6pt}
        \item Sharp, well-dated discontinuity --- ideal for causal designs
        \item Single treated unit $\Rightarrow$ DiD parallel-trends is hard
              to defend; SCM is natural choice
        \item Multiple competing channels (safety, fiscal, economic)
              $\Rightarrow$ direction of effect is empirical
        \item Policy still in force a decade later --- post-period covers
              7+ years with rich variation (incl.\ COVID-19 shock)
      \end{itemize}

    \column{0.40\textwidth}
      \begin{tcolorbox}[statbox,width=\linewidth]
        \fontsize{28}{32}\selectfont\bfseries\color{AmberAccent}120M\\
        \small\color{white}affected population
      \end{tcolorbox}
      \vspace{0.2cm}
      \begin{tcolorbox}[statbox,width=\linewidth]
        \fontsize{28}{32}\selectfont\bfseries\color{AmberAccent}$\sim$\rupee4{,}000\,Cr\\
        \small\color{white}pre-prohibition state excise (FY15--16)
      \end{tcolorbox}
      \vspace{0.2cm}
      \begin{tcolorbox}[statbox,width=\linewidth]
        \fontsize{28}{32}\selectfont\bfseries\color{AmberAccent}3 outcomes\\
        \small\color{white}safety / fiscal / economic
      \end{tcolorbox}
  \end{columns}
\end{frame}

% ── SLIDE 4: Research Questions ─────────────────────────────────────────────
\begin{frame}{Research Questions}
  \vspace{0.4cm}
  \begin{tcolorbox}[navybox,title={\faQuestion\enspace Three questions, one identification strategy}]
    \begin{enumerate}\setlength\itemsep{8pt}
      \item Did prohibition \highlight{reduce road-accident deaths} relative to
            a counterfactual Bihar that did not enact prohibition?
      \item Did prohibition \highlight{reduce state own-tax revenue} via the
            loss of the excise component?
      \item Did prohibition \highlight{accelerate or depress per-capita
            economic growth} relative to comparable states?
    \end{enumerate}
  \end{tcolorbox}

  \vspace{0.5cm}
  \textbf{\navy{Identification strategy.}} Construct a \emph{synthetic Bihar}
  as a weighted average of donor states whose pre-2016 trajectory tracks
  Bihar's, then read the post-2016 gap as the treatment effect. Triangulate
  with a Bayesian structural counterfactual (BSTS).

  \vspace{0.4cm}
  \begin{tcolorbox}[greenbox]
    \small\textbf{Why three outcomes?} Each isolates a distinct welfare
    dimension --- Q1 captures the \greentext{intended public-health} gain,
    Q2 the \redtext{direct fiscal cost}, Q3 the \emph{net} economic effect
    of competing channels.
  \end{tcolorbox}
\end{frame}

% ── SLIDE 5: Identification Challenge ───────────────────────────────────────
\begin{frame}{The Identification Challenge}
  \vspace{0.2cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \textbf{\navy{Standard tools fall short.}}
      \begin{itemize}\setlength\itemsep{5pt}
        \item \redtext{DiD} requires parallel pre-trends, but Bihar is an
              outlier on income, urbanisation, and literacy
        \item \redtext{RDD} needs a running variable; the timing
              discontinuity gives one but only at the within-Bihar level
        \item \redtext{IV} has no obvious instrument for state-wide policy
        \item Single treated unit $\Rightarrow$ no within-treatment variation
      \end{itemize}
      \vspace{0.3cm}
      \begin{tcolorbox}[amberbox]
        \small \textbf{SCM matches the structure.} \highlight{One treated
        unit, many donors, long pre-period} --- exactly the setting Abadie,
        Diamond \& Hainmueller (2010) introduced the method for.
      \end{tcolorbox}

    \column{0.48\textwidth}
      \begin{tcolorbox}[navybox,title={\faExclamationTriangle\enspace Convex hull caveat (Abadie 2021)}]
        \small Bihar's pre-period income and urbanisation lie at the lower
        edge of the donor pool. A naive SCM optimiser collapses to a
        \redtext{corner solution} (100\% weight on Uttar Pradesh).
        \vspace{0.18cm}\\
        \emph{Fix:} include lagged outcomes as
        \texttt{special\_predictors} (Abadie 2010 §IV) --- this anchors
        the synthetic to Bihar's actual trajectory, not just its
        socio-economic profile.
      \end{tcolorbox}
  \end{columns}
\end{frame}

% ── SLIDE 8: SCM Method ─────────────────────────────────────────────────────
\begin{frame}{Method --- Synthetic Control}
  \vspace{0.1cm}
  Treated unit $i=1$ (Bihar) and $J$ donors. Choose weights
  $\mathbf{w}=(w_2,\dots,w_{J+1})^\top$ to solve:

  \vspace{0.1cm}
  \begin{tcolorbox}[navybox]
    \centering
    $\displaystyle
      \mathbf{w}^* \;=\; \arg\min_{\mathbf{w}\in\mathcal{W}}\;
      \bigl\| X_1 - X_0\,\mathbf{w} \bigr\|_V \,,\quad
      \mathcal{W}=\Bigl\{\mathbf{w}\!:\!w_j\ge 0,\;\sum w_j=1\Bigr\}$
  \end{tcolorbox}

  \vspace{0.15cm}
  \footnotesize
  $X_1$: predictor vector for treated; $X_0$: $K\times J$ donor matrix; $V$:
  predictor-importance weight matrix (Abadie--Diamond--Hainmueller estimate
  $V$ to minimise pre-period MSPE).

  \vspace{0.35cm}
  \textbf{\navy{Two consequences of the convex-hull constraint:}}
  \begin{itemize}\setlength\itemsep{4pt}\small
    \item Synthetic Bihar cannot extrapolate beyond the donor envelope ---
          good for honest counterfactuals
    \item If Bihar lies near a corner, weights collapse to a single donor
          $\Rightarrow$ poor pre-period fit + spurious post-period gap
  \end{itemize}

  \vspace{0.25cm}
  \begin{tcolorbox}[greenbox]
    \scriptsize\textbf{Inference.} \textbf{In-space placebo} (Abadie et al.\
    2010): refit SCM treating each donor as if it were treated. Report
    Bihar's post/pre RMSPE \emph{ratio rank} relative to the placebo
    distribution; permutation $p$-value = rank / $(J+1)$.
  \end{tcolorbox}
\end{frame}

% ── SLIDE 9: Convex Hull Problem ────────────────────────────────────────────
\begin{frame}{The Convex Hull Problem --- And Its Fix}
  \vspace{0.1cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \centering
      \fig[\linewidth]{eda_bihar_vs_donor_avg.png}\\[2pt]
      \scriptsize\color{GrayMid} Bihar lies persistently \emph{below} the
      donor-pool average level on road-accident deaths.

    \column{0.46\textwidth}
      \textbf{\navy{Without the fix.}}
      \begin{itemize}\setlength\itemsep{3pt}\small
        \item Match on socio-economic predictors only
        \item Optimiser finds 100\% UP weight (closest by income)
        \item Pre-period RMSPE \redtext{$\sim$1{,}300 deaths} (huge)
        \item Post-period gap meaningless --- pre-trend doesn't match
      \end{itemize}
      \vspace{0.25cm}
      \textbf{\navy{With lagged-outcome anchoring (Abadie 2010 §IV).}}
      \begin{itemize}\setlength\itemsep{3pt}\small
        \item Add each pre-treatment year's outcome as a
              \texttt{special\_predictor}
        \item Forces synthetic to track Bihar's actual trajectory
        \item Pre-period RMSPE drops to \greentext{42 deaths} ($<$\,1\%
              of Bihar's mean) --- excellent fit
      \end{itemize}
  \end{columns}
\end{frame}

% ── SLIDE 14: BSTS Method ───────────────────────────────────────────────────
\begin{frame}{Method --- Bayesian Structural Time Series}
  \vspace{0.05cm}
  Brodersen et al.\ (2015): observed series decomposes as

  \vspace{0.05cm}
  \begin{tcolorbox}[navybox]
    \centering
    $y_t \;=\; \mu_t \;+\; \tau_t \;+\; \boldsymbol{\beta}^\top
              \mathbf{x}_t \;+\; \varepsilon_t$
  \end{tcolorbox}

  \vspace{0.1cm}
  \footnotesize $\mu_t$: local-level trend $\;|\;$
  $\tau_t$: seasonality (omitted, annual data) $\;|\;$
  $\boldsymbol{\beta}^\top\mathbf{x}_t$: regression on donor outcomes
  with \emph{spike-and-slab} priors $\;|\;$
  $\varepsilon_t \sim \mathcal{N}(0,\sigma^2)$.

  \vspace{0.4cm}
  \textbf{\navy{Why BSTS triangulates with SCM.}}
  \begin{itemize}\setlength\itemsep{4pt}\small
    \item Different functional form (linear regression with priors vs.\
          convex average) $\Rightarrow$ different mis-specification risks
    \item Yields full posterior $\Rightarrow$ credible intervals on the
          counterfactual, not just point estimate
    \item Spike-and-slab automatically prunes irrelevant donors
  \end{itemize}

  \vspace{0.25cm}
  \begin{tcolorbox}[redbox]
    \scriptsize\textbf{Caveat for short pre-period.} With $n_\text{pre}=6$
    observations and $J=13$ donors, the regression component
    \redtext{overfits}. We restrict to the top-$k$ donors by pre-period
    correlation (\texttt{max\_donors=3}) per Brodersen §3.2.
  \end{tcolorbox}
\end{frame}

% ── SLIDE 22: Limitations ───────────────────────────────────────────────────
\begin{frame}{Limitations}
  \vspace{0.15cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{tcolorbox}[navybox,title={\faExclamationCircle\enspace Data}]
        \scriptsize\setlength\itemsep{2pt}
        \begin{itemize}
          \item NSDP T19 starts FY 2011--12 $\Rightarrow$ only 4 pre-period
                years in matching window
          \item Census 2021 delayed; urban share \& literacy
                \emph{interpolated} from 2011 anchor
          \item RBI State Finances excise sub-head WAF-blocked
                $\Rightarrow$ composite own-tax used as proxy
        \end{itemize}
      \end{tcolorbox}
      \vspace{0.25cm}
      \begin{tcolorbox}[navybox,title={\faExclamationCircle\enspace Identification}]
        \scriptsize\setlength\itemsep{2pt}
        \begin{itemize}
          \item Bihar near edge of donor convex hull on income/urbanisation
          \item COVID-19 (2020--21) is a confounding shock for all
                post-2020 estimates
          \item Inter-state migration post-2016 not captured
        \end{itemize}
      \end{tcolorbox}

    \column{0.48\textwidth}
      \begin{tcolorbox}[navybox,title={\faExclamationCircle\enspace Method}]
        \scriptsize\setlength\itemsep{2pt}
        \begin{itemize}
          \item BSTS overfits with $n_\text{pre}=6 < J=13$
                $\Rightarrow$ \texttt{max\_donors=3} restriction (subjective)
          \item Growth BSTS uses single donor + shifted treatment date
                (2017) --- credible interval reflects model
                constraint, not posterior uncertainty
          \item Permutation $p$-values from $J=14$ are coarse
                ($1/14 \approx 0.07$ minimum)
        \end{itemize}
      \end{tcolorbox}
      \vspace{0.25cm}
      \begin{tcolorbox}[amberbox]
        \scriptsize \highlight{Honest framing:} Bihar is a hard SCM case
        --- short pre-period, outlier on covariates, COVID confounder.
        Triangulating SCM and BSTS is the strongest defence.
      \end{tcolorbox}
  \end{columns}
\end{frame}

% ── SLIDE 24: Conclusions ───────────────────────────────────────────────────
\begin{frame}{Conclusions}
  \vspace{0.2cm}
  \begin{tcolorbox}[navybox,title={\faAward\enspace Headline}]
    \small
    Bihar's 2016 prohibition produced a \greentext{statistically meaningful
    short-run reduction in road-accident deaths} (rank 2 / 14, $p\approx
    0.071$) that \redtext{reversed by 2018}, alongside a
    \redtext{persistent fiscal contraction} ($-$\rupee2{,}017 Cr/yr, attenuated
    composite) and a \emph{suggestive but not significant} growth headwind
    ($-3$\,pp/yr, $p\approx 0.43$).
  \end{tcolorbox}

  \vspace{0.4cm}
  \textbf{\navy{Methodological takeaways.}}
  \begin{itemize}\setlength\itemsep{4pt}\small
    \item Lagged-outcome anchoring is essential for treated units near the
          convex hull edge --- without it the SCM degenerates
    \item BSTS and SCM disagree on magnitude but agree on direction across
          all three outcomes --- triangulation strengthens the qualitative
          claim, not the precise number
    \item For short pre-periods, sparser BSTS specifications (1--3 donors)
          dominate the full-pool default
  \end{itemize}

  \vspace{0.35cm}
  \begin{tcolorbox}[amberbox]
    \scriptsize \textbf{Future work.} Pure-excise series via state-budget
    archive scrape; high-frequency outcomes (monthly Vahan data); spatial
    spillover model with Jharkhand / UP / WB borders.
  \end{tcolorbox}
\end{frame}
